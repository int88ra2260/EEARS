<script>
const SEGMENTS = [
  { id: 'dive_01', duration: 8, section: 0, isDive: true  },
  { id: 'conn_01', duration: 5, section: null, isDive: false },
  { id: 'dive_02', duration: 8, section: 1, isDive: true  },
  { id: 'conn_02', duration: 5, section: null, isDive: false },
  { id: 'dive_03', duration: 8, section: 2, isDive: true  },
  { id: 'conn_03', duration: 5, section: null, isDive: false },
  { id: 'dive_04', duration: 8, section: 3, isDive: true  },
];

const TOTAL = SEGMENTS.reduce((s, x) => s + x.duration, 0);
let accum = 0;
SEGMENTS.forEach(seg => {
  seg.startP = accum / TOTAL;
  accum += seg.duration;
  seg.endP = accum / TOTAL;
});

// Caption range: show late in dive, hold through half of connector
SEGMENTS.forEach((seg, i) => {
  if (!seg.isDive) return;
  const span = seg.endP - seg.startP;
  const nextConn = SEGMENTS[i + 1];
  seg.captionRange = [
    seg.startP + span * 0.55,
    nextConn ? nextConn.startP + (nextConn.endP - nextConn.startP) * 0.5 : seg.endP
  ];
});

const world = document.getElementById('world');
const stages = document.querySelectorAll('.stage');
const captions = [0,1,2,3].map(i => document.getElementById('cap-' + i));
const dots = document.querySelectorAll('.dot');
const progressBar = document.getElementById('progress');

const VH_PER_SEGMENT = 220;
world.style.height = (SEGMENTS.length * VH_PER_SEGMENT) + 'vh';

let ticking = false;
function update() {
  const rect = world.getBoundingClientRect();
  const worldH = world.offsetHeight - window.innerHeight;
  const scrolled = Math.max(0, -rect.top);
  const p = Math.min(1, scrolled / worldH);
  progressBar.style.width = (p * 100) + '%';

  // find active segment
  let activeIdx = SEGMENTS.length - 1;
  for (let i = 0; i < SEGMENTS.length; i++) {
    if (p <= SEGMENTS[i].endP) { activeIdx = i; break; }
  }
  const active = SEGMENTS[activeIdx];
  const localP = Math.max(0, Math.min(1, (p - active.startP) / (active.endP - active.startP)));

  stages.forEach((stage, i) => {
    const isActive = i === activeIdx;
    stage.classList.toggle('active', isActive);
    if (!isActive) return;

    const seg = SEGMENTS[i];
    if (seg.isDive) {
      // Dive camera: high & small → close & large
      const zoom = 0.75 + localP * 1.8;   // 0.75 → 2.55
      const ty = 20 - localP * 90;         // downward push
      stage.style.setProperty('--zoom', zoom);
      stage.style.setProperty('--tx', '0px');
      stage.style.setProperty('--ty', ty + 'px');

      // Roof lift after 45% progress (if scene has a roof)
      const hasRoof = stage.dataset.hasroof === '1';
      if (hasRoof) {
        const roofP = Math.max(0, Math.min(1, (localP - 0.45) / 0.4));
        const roof = stage.querySelector('.roof');
        if (roof) {
          roof.style.setProperty('--roof-y', (-90 * roofP) + 'px');
          roof.style.setProperty('--roof-op', 1 - roofP);
        }
      }
    } else {
      // Connector: aerial pan left → right, arcs high in middle
      const tx = 220 - localP * 440;
      const zoom = 1.0 - Math.abs(localP - 0.5) * 0.35;
      stage.style.setProperty('--zoom', zoom);
      stage.style.setProperty('--tx', tx + 'px');
      stage.style.setProperty('--ty', '0px');
    }
  });

  captions.forEach((cap, sectionIdx) => {
    const diveSeg = SEGMENTS.find(s => s.section === sectionIdx);
    if (!diveSeg) return;
    const [start, end] = diveSeg.captionRange;
    cap.classList.toggle('show', p >= start && p <= end);
  });

  const currentSection = active.section !== null
    ? active.section
    : (activeIdx > 0 ? SEGMENTS[activeIdx - 1].section : 0);
  dots.forEach((dot, i) => dot.classList.toggle('active', i === currentSection));

  ticking = false;
}

function onScroll() {
  if (!ticking) { requestAnimationFrame(update); ticking = true; }
}

dots.forEach((dot, i) => {
  dot.addEventListener('click', () => {
    const diveSeg = SEGMENTS.find(s => s.section === i);
    const worldH = world.offsetHeight - window.innerHeight;
    window.scrollTo({ top: (diveSeg.startP + 0.03) * worldH, behavior: 'smooth' });
  });
});

window.addEventListener('scroll', onScroll, { passive: true });
window.addEventListener('resize', onScroll);
update();

// Hide hint after user scrolls
window.addEventListener('scroll', () => {
  const hint = document.querySelector('.hint');
  if (hint && window.scrollY > 200) hint.style.opacity = '0';
}, { once: false, passive: true });
</script>
